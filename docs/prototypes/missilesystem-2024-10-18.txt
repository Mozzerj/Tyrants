createDialog "VLSTargetMenu";
params["_Asset"];

_targetPos = [2000,21000,0];
_VLS = (_Asset # 0);

setMissileDir = {
    params["_yaw","_pitch","_roll"];

    _missile setVectorDirAndUp [
        [sin _yaw * cos _pitch, cos _yaw * cos _pitch, sin _pitch],
        [[sin _roll, -sin _pitch, cos _roll * cos _pitch], -_yaw] call BIS_fnc_rotateVector2D
    ];
};


missileWaypointCalculation = {
    params["_missile", "_targetPos"];

    _LaunchSite = getPos _VLS;
    
    _plots = [];

    _pointsCount = (_LaunchSite distance2D _targetPos);

    _x1 = _LaunchSite # 0;
    _y1 = _LaunchSite # 1;
    _x2 = _targetPos # 0;
    _y2 = _targetPos # 1;

    for "_i" from 0 to (_pointsCount - 1) do {
        _t = _i / (_pointsCount - 1);
        _x = (1 - _t) * _x1 + _t * _x2;
        _y = (1 - _t) * _y1 + _t * _y2;
        _plots pushBack [_x, _y];
    };

    {
        if (_forEachIndex % 150 == 0) then {
            _plotHeight = createVehicle ["ArtilleryTargetW", _x, [], 0, "CAN_COLLIDE"];
            (_plots # _forEachIndex) pushBack (getPos _plotHeight # 2);
            deleteVehicle _plotHeight;
        };
    } forEach _plots;

    _HighPlot = [0,0,-100];
    _MainPlots = [];

    {
        if (_forEachIndex % 500 == 0) then {
            _MainPlots pushBack _HighPlot;
            _HighPlot = [0,0,-100];
        };

        if (_x # 2 > _HighPlot # 2) then {_HighPlot = _x};

    } forEach _plots;

    _MainPlots;
};


missileCruiseSystem = {
    params["_missile", "_target", "_Path"];


    _minHeightATL = 25;

    for "_i" from 0 to 3 do {_path deleteAt 0;};

    _markers = [];
    _irName = 0;

    {
        // Create a random position near the waypoint and add a marker
        _WaypointPos = _x vectorAdd [random [-25,0,25], random [-25,0,25], 60];
        private _markerName = createMarker [format ["missileMarker_%1", _x], _WaypointPos];
        _markerName setMarkerType "mil_dot";
        _markerName setMarkerColor "ColorRed";
        _markers pushBack _markerName;

        while {(getPos _missile distance _WaypointPos) > 100 && alive _missile} do {

            _dir = (getPos _missile) getDir (_WaypointPos);

            _diff = _WaypointPos vectorDiff getPos _missile;
            _pitch = (_diff select 2) atan2 (sqrt ((_diff select 0)^2 + (_diff select 1)^2));

            [_dir, _pitch, 0] call setMissileDir;

            sleep 0.1;
        };

    } forEach _Path;

    {
        deleteMarker _x;
    } forEach _markers;


    deleteVehicle _FX_smoke_source;
};




fireMissile = {
    params["_VLS", "_targetPos", "_Path"];

    _target = createVehicle ["ArtilleryTargetW", _targetPos, [], 0, "CAN_COLLIDE"];
    _missile = createVehicle ["ammo_Missile_Cruise_01", (getPos _VLS) vectorAdd [0,0,2], [], 0, "CAN_COLLIDE"];

    _missile setVelocity [0, 0, 7];

    _FX_smoke_source = "#particlesource" createVehicleLocal (getpos _missile);
    _FX_smoke_source setParticleClass "Missile5";
    _FX_smoke_source attachto [_missile,[0,-1,0]];

    _dir = getPos _missile getDir _targetPos;

    _dir = _dir + random [-10,0,10];

    [_dir, 90, 0] call setMissileDir;

    sleep 2;

    for "_i" from 0 to 85 do {

        
        [_dir, 90 - _i, 0] call setMissileDir;

        sleep 0.02;
        
    };

    sleep 1;

    [_missile,_target, _Path] call missileCruiseSystem;

};

for "_i" from 0 to 0 do {

    _Waypoints = [_VLS,_targetPos] call missileWaypointCalculation;
    [_VLS,_targetPos, _Waypoints] call fireMissile;

};
